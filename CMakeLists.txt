cmake_minimum_required(VERSION 4.0)
project(ALP8BV2 CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)          # for clangd
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)            # breaks autogenerated Verilator headers

# --- Dependencies -----------------------------------------------------------

find_package(verilator REQUIRED)
find_package(SDL3 REQUIRED)

include(FetchContent)
FetchContent_Declare(
    magic_enum
    GIT_REPOSITORY https://github.com/Neargye/magic_enum.git
    GIT_TAG        v0.9.8
    GIT_SHALLOW    TRUE
    SYSTEM
)
FetchContent_MakeAvailable(magic_enum)

# --- Shared: .alp8b binary file format ---------------------------------------
# INTERFACE target: compiles no sources, only exports shared/ as an include
# dir to every target that links it (assembler writes, sim reads)

add_library(alp8b_file INTERFACE)
target_include_directories(alp8b_file INTERFACE "${CMAKE_SOURCE_DIR}/shared")

# --- Helper: build a Verilator testbench executable -------------------------

function(add_hardware_test TARGET_NAME TOP_MODULE CPP_TB VERILOG_SOURCES VERILOG_INCLUDES)
    # create a separate library target for the Verilated model
    set(SIM_LIB_TARGET "${TARGET_NAME}_sim")
    add_library(${SIM_LIB_TARGET} STATIC)

    # disable all compiler warnings on generated Verilator C++ files
    # since we cant deal with warnings on autogenerated files
    target_compile_options(${SIM_LIB_TARGET} PRIVATE -w)

    verilate(${SIM_LIB_TARGET}
        PREFIX          "V${TOP_MODULE}"
        TOP_MODULE      "${TOP_MODULE}"
        SOURCES         ${VERILOG_SOURCES}
        INCLUDE_DIRS    ${VERILOG_INCLUDES}
        VERILATOR_ARGS  "-Wall"
    )

    # Force all includes exported by the simulation model to be SYSTEM headers
    set_target_properties(${SIM_LIB_TARGET} PROPERTIES
        INTERFACE_SYSTEM_INCLUDE_DIRECTORIES "$<TARGET_PROPERTY:${SIM_LIB_TARGET},INTERFACE_INCLUDE_DIRECTORIES>"
    )

    # create testbench executable
    add_executable(${TARGET_NAME} "${CPP_TB}")
    target_include_directories(${TARGET_NAME} PRIVATE sim/include)
    # warnings on ONLY the testbench code
    target_compile_options(${TARGET_NAME} PRIVATE -Wall -Wextra)

    # link together
    target_link_libraries(${TARGET_NAME} PRIVATE
        ${SIM_LIB_TARGET}
        magic_enum::magic_enum
    )
endfunction()

# --- Testbenches --------------------------------------------------------

add_hardware_test(Vtbsim_alu alu
    "sim/src/test/tb_alu.cpp"
    "src/dp/alu/alu.sv;src/dp/alu/alu_defs.sv"
    "src/dp/alu"
)

add_hardware_test(Vtbsim_ram ram
    "sim/src/test/tb_ram.cpp"
    "src/dp/ram.sv"
    "src/dp"
)

# --- Top-level executable: ALP8BV2 ------------------------------------------

set(ALP8B alp8bv2)

# static lib (to generate without warning pollution)
set(ALP8B_SIM_LIB "${ALP8B}_sim")
add_library(${ALP8B_SIM_LIB} STATIC)

target_compile_options(${ALP8B_SIM_LIB} PRIVATE -w)

verilate(${ALP8B_SIM_LIB}
    PREFIX          Valp8b
    TOP_MODULE      cpu_top
    SOURCES         src/cpu_top.sv
    INCLUDE_DIRS    "src/dp;src/cu;src/dp/alu"
    VERILATOR_ARGS  "-Wall"
)

set_target_properties(${ALP8B_SIM_LIB} PROPERTIES
    INTERFACE_SYSTEM_INCLUDE_DIRECTORIES "$<TARGET_PROPERTY:${ALP8B_SIM_LIB},INTERFACE_INCLUDE_DIRECTORIES>"
)

# create the main program exec
add_executable(${ALP8B} sim/src/cpu.cpp)
target_include_directories(${ALP8B} PRIVATE sim/include)
target_compile_options(${ALP8B} PRIVATE -Wall -Wextra)

# link the simlib & others
target_link_libraries(${ALP8B} PRIVATE magic_enum::magic_enum SDL3::SDL3 alp8b_file ${ALP8B_SIM_LIB})
